	<script>
	(function($){
		var $input   = $('#pv4-search');
		var $results = $('#pv4-search-results');
		var timer    = null;
		var xhr      = null;
		var lastTerm = '';

		function escapeHtml(str){
			return String(str).replace(/[&<>"']/g, function(c){
				return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
			});
		}

		var $wrap = $('#pv4-search-wrap');

		// Chargement : l'élément englobant porte la classe is-loading (l'habillage affiche l'indicateur) et aria-busy
		function setLoading(on){
			$wrap.toggleClass('is-loading', on).attr('aria-busy', on ? 'true' : 'false');
		}

		function message(text){
			$results.empty().append($('<li class="menu-search-empty">').text(text)).show();
		}

		function hideResults(){
			if(xhr){ xhr.abort(); xhr = null; }
			setLoading(false);
			$results.empty().hide();
		}

		function renderResults(items){
			$results.empty();
			if(!items.length){
				$results.append('<li class="menu-search-empty">Aucun match trouvé</li>');
			}else{
				items.forEach(function(item){
					$results.append(
						'<li><a href="' + escapeHtml(item.url) + '">' +
							'<span class="menu-search-logos">' +
								'<img src="' + escapeHtml(item.home_logo) + '" alt="" loading="lazy" onerror="this.style.visibility=\'hidden\'">' +
								'<img src="' + escapeHtml(item.away_logo) + '" alt="" loading="lazy" onerror="this.style.visibility=\'hidden\'">' +
							'</span>' +
							'<span class="menu-search-infos">' +
								'<span class="menu-search-name">' + escapeHtml(item.event_name) + '</span>' +
								'<small>' + (item.sport_name ? '<span class="menu-search-sport">' + escapeHtml(item.sport_name) + '</span>' : '') + escapeHtml(item.stage_name) + ' &middot; ' + escapeHtml(item.startdate_fr) + '</small>' +
							'</span>' +
						'</a></li>'
					);
				});
			}
			$results.show();
		}

		function search(term){
			if(xhr) xhr.abort();
			setLoading(true);
			var req = xhr = $.ajax({
				url: '/?option=com_pronocote&view=event&format=ajax&layout=getEventsByQuery',
				type: 'GET',
				dataType: 'json',
				data: { q: term }
			});
			req.done(function(response){
				if(xhr !== req) return;
				if(response && response.throttled){ // limite de fréquence côté serveur (20 requêtes / 10 s / IP)
					message('Trop de recherches : réessayez dans quelques secondes.');
					lastTerm = '';
					return;
				}
				renderResults((response && response.results) || []);
			}).fail(function(jq, status){
				if(status === 'abort' || xhr !== req) return; // remplacée par une recherche plus récente
				message('Recherche indisponible, réessayez.');
				lastTerm = '';
			}).always(function(){
				if(xhr === req){ xhr = null; setLoading(false); }
			});
		}

		$input.on('input', function(){
			var term = $input.val().trim();
			clearTimeout(timer);

			if(term.length < 3){
				hideResults();
				lastTerm = '';
				return;
			}
			if(term === lastTerm) return;

			timer = setTimeout(function(){
				lastTerm = term;
				search(term);
			}, 300);
		});

		$(document).on('click', function(e){
			if(!$(e.target).closest('#pv4-search-wrap').length){
				hideResults();
			}
		});

		$input.on('keydown', function(e){
			if(e.which === 27){ // Echap
				hideResults();
			}
		});
	})(jQuery);

<h3>Comment battre les bookmakers sur vos pronostics foot ?</h3>
<p>A chaque match de foot, correspond des cotes, qui correspondent elles-mêmes à des probabilités. Exemple, <strong>sur les pronostics foot 1N2, il existe 3 issues</strong> : la victoire à domicile, le match nul et la victoire à l'extérieur.&nbsp;Pour trouver des <strong>Value Bet</strong>, c'est à dire des <strong>paris football qui vaillent le coup</strong>, il faut que la probalitié que vous donnez à l'une des issues soit plus importante que celle donnée par le bookmaker</p>
<p>Exemple : le PSG affronte l'OM au Vélodrome et les sites de paris sportifs français offrent une cote de 2,00 au PSG (probabilité de 50%), d'un match nul à 3,90 (probabilité de 25%) et une cote de 3,90 pour l'OM (probabilité de 25%).&nbsp;</p>
<ul>
<li>Si vous estimez que la probabilité du PSG de gagner est plutôt de 50%, alors le Value Bet est sur les Parisiens.</li>
<li>Si vous estimez qu'un match nul est très possible, la probabilité de&nbsp;25% offerte par la cote doit vous intéresser.</li>
<li>Si vous estimez que l'OM a + de 25% de l'emporter, alors le Value Bet est sur les Marseillais.</li>
</ul>
